│   │   ├── tutor.py                    # GET/PUT /tutor/profile, /subjects, /certificates
│   │   ├── schedule.py                 # POST /schedule, POST /booking, GET /schedule/<tutor_id>
│   │   ├── payment.py                  # POST /payment, GET /payment/history
│   │   ├── review.py                   # POST /review, GET /tutor/<id>/reviews
│   │   └── admin.py                    # PATCH /admin/verify/<id>, GET /admin/dashboard
│   │
│   └── utils/
